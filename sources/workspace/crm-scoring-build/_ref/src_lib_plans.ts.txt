/** Percent saved compared with the GoHighLevel list price (rounded down). */
export function savingsPercent(ours: number, theirs: number): number {
  if (theirs <= 0) return 0;
  return Math.floor(((theirs - ours) / theirs) * 100);
}

export function planById(id: string | null | undefined): Plan {
  return PLANS.find((p) => p.id === id) ?? PLANS[0];
}

export function hasFeature(features: readonly string[] | null | undefined, f: Feature): boolean {
  return Boolean(features?.includes(f));
}

/** Cheapest paid plan that includes a feature (for "Upgrade to …" hints). */
export function planFor(f: Feature): Plan | undefined {
  return PLANS.filter((p) => p.billable).find((p) => p.features.includes(f));
}
